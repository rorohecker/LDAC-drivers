cmake_minimum_required(VERSION 3.20)
project(xm5_ldac_driver LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(XM5_ASAN "Enable AddressSanitizer for user-mode targets" OFF)
if(XM5_ASAN)
 if(MSVC)
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/INCREMENTAL:NO /DEBUG)
 else()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
 endif()
endif()
add_library(ldac_encoder STATIC vendor/libldac/src/ldaclib.c vendor/libldac/src/ldacBT.c)
target_include_directories(ldac_encoder PUBLIC vendor/libldac/inc)
if(MSVC)
 target_compile_definitions(ldac_encoder PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
 target_link_libraries(ldac_encoder PUBLIC m)
endif()
add_library(xm5_core STATIC core/avdtp.cpp core/encoder.cpp)
target_link_libraries(xm5_core PUBLIC ldac_encoder)
target_include_directories(xm5_core PUBLIC core)
add_executable(ldac_encode tools/encode.c)
target_link_libraries(ldac_encode PRIVATE ldac_encoder)
if(MSVC)
 target_compile_definitions(ldac_encode PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core PRIVATE xm5_core)
enable_testing()
add_test(NAME protocol_and_encoder COMMAND test_core)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME offline_encoder COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_encode.py $<TARGET_FILE:ldac_encode>)
endif()
if(WIN32)
 add_executable(xm5_stream windows/main.cpp windows/transport.cpp windows/capture.cpp)
 target_compile_definitions(xm5_stream PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
 target_link_libraries(xm5_stream PRIVATE xm5_core setupapi ole32 uuid avrt)
 add_test(NAME client_help COMMAND xm5_stream --help)
 add_executable(test_pacer tests/test_pacer.cpp)
 target_compile_definitions(test_pacer PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
 target_link_libraries(test_pacer PRIVATE avrt)
 add_test(NAME windows_pacer COMMAND test_pacer)
 execute_process(COMMAND powershell.exe -NoProfile -Command Get-ExecutionPolicy
  OUTPUT_VARIABLE XM5_PS_POLICY OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
 if(XM5_PS_POLICY STREQUAL "RemoteSigned" OR XM5_PS_POLICY STREQUAL "Unrestricted")
  add_test(NAME powershell_tools COMMAND powershell.exe -NoProfile -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_tools.ps1)
 else()
  message(STATUS "PowerShell script tests skipped under the current execution policy: ${XM5_PS_POLICY}")
 endif()
endif()
